# Optional Apache / cPanel settings for the Brand Image website.
#
# IMPORTANT: your hosting account may already have an .htaccess file (for example
# for the /outdoor section). Do NOT overwrite it — copy only the blocks you need
# into the existing file.

# Never serve hidden files such as .env (the contact form's SMTP settings belong one folder
# ABOVE public_html; this is a safety net in case one is uploaded into it by mistake)
<FilesMatch "^\.(env|git)">
  Require all denied
</FilesMatch>

# Show the branded "page not found" page
ErrorDocument 404 /404.html

# Compress text files
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml application/xml text/plain application/manifest+json
</IfModule>

# Cache static assets (CSS/JS file names change on every build via ?v=…)
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/html "access plus 0 seconds"
  ExpiresByType text/css "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType image/webp "access plus 6 months"
  ExpiresByType image/png "access plus 6 months"
  ExpiresByType image/jpeg "access plus 6 months"
  ExpiresByType image/x-icon "access plus 6 months"
  ExpiresByType font/woff2 "access plus 1 year"
</IfModule>

# Correct type for the web manifest
AddType application/manifest+json .webmanifest
AddType font/woff2 .woff2
